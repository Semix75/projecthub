✅ Étape 2 : API et Gestion des Conversations
🔹 Pourquoi une API est utile ?
Une API permet de :

Récupérer les conversations et les messages associés.
Créer une conversation entre deux utilisateurs ou un groupe.
Envoyer un message et le diffuser en temps réel via Mercure.
🔹 Routes API nécessaires :
GET /api/conversations → Récupérer les conversations d'un utilisateur.
GET /api/conversations/{id} → Récupérer les messages d'une conversation.
POST /api/conversations → Créer une conversation.
POST /api/messages → Envoyer un message.

import React, { useEffect, useState } from 'react';
import { useParams } from 'react-router-dom';

const ChatView = ({ conversationId, token }) => {
    const [messages, setMessages] = useState([]);
    const [content, setContent] = useState('');
    const [user, setUser] = useState(null);

    // Récupère l'utilisateur courant
    useEffect(() => {
        fetch('/api/me', {
            headers: {
                Authorization: `Bearer ${token}`,
            },
        })
        .then(res => res.json())
        .then(data => setUser(data))
        .catch(err => console.error('Erreur récupération user', err));
    }, [token]);

    // Récupère les messages de la conversation
    useEffect(() => {
        fetch(`/api/conversations/${conversationId}/messages`)
          .then(res => res.json())
          .then(data => setMessages(data))
          .catch(err => console.error(err));
      }, [conversationId]);
    // Abonnement à Mercure
    useEffect(() => {
        const url = new URL('http://localhost:3000/.well-known/mercure');
        url.searchParams.append('topic', `/conversations/${conversationId}`);

        const eventSource = new EventSource(url, { withCredentials: true });

        eventSource.onmessage = (event) => {
            const data = JSON.parse(event.data);
            setMessages((prevMessages) => [...prevMessages, data]);
        };

        return () => {
            eventSource.close();
        };
    }, [conversationId]);

    // Envoi du message
    const handleSubmit = async (e) => {
        e.preventDefault();
        if (!content.trim()) return;

        await fetch('/api/messages', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                Authorization: `Bearer ${token}`,
            },
            body: JSON.stringify({
                content,
                conversation: `/api/conversations/${conversationId}`,
            }),
        });

        setContent('');
    };

    return (
        <div>
            <h2>Conversation #{conversationId}</h2>
            <ul>
                {messages.map((msg) => (
                    <li key={msg.id}>
                        <strong>{msg.sendBy?.username}:</strong> {msg.content}
                    </li>
                ))}
            </ul>

            <form onSubmit={handleSubmit}>
                <input
                    type="text"
                    value={content}
                    onChange={(e) => setContent(e.target.value)}
                    placeholder="Votre message..."
                />
                <button type="submit">Envoyer</button>
            </form>
        </div>
    );
};

export default ChatView;
